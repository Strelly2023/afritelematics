   226	def _load_signing_keyring(*, require_rotation: bool) -> tuple[dict[str, bytes], str]:
   227	    encoded_registry = os.getenv("NOVAID_SIGNING_KEYS_JSON", "")
   228	    active_key_id = os.getenv("NOVAID_ACTIVE_SIGNING_KEY_ID", "")
   229	    if encoded_registry:
   230	        try:
   231	            parsed = json.loads(encoded_registry)
   232	        except json.JSONDecodeError as exc:
   233	            raise RuntimeError("invalid_novaid_signing_key_registry") from exc
   234	        if not isinstance(parsed, dict) or not parsed:
   235	            raise RuntimeError("invalid_novaid_signing_key_registry")
   236	        keys = {
   237	            str(key_id): str(secret).encode()
   238	            for key_id, secret in parsed.items()
   239	        }
   240	        if (
   241	            not active_key_id
   242	            or active_key_id not in keys
   243	            or any(len(secret) < 32 for secret in keys.values())
   244	        ):
   245	            raise RuntimeError("invalid_novaid_signing_key_registry")
   246	        return keys, active_key_id
   247	    if require_rotation:
   248	        raise RuntimeError("missing_novaid_signing_key_registry")
   249	    legacy_key = os.getenv(
   250	        "NOVAID_SIGNING_KEY",
   251	        "development-only-signing-key-00001",
   252	    ).encode()
   253	    if len(legacy_key) < 32:
   254	        raise RuntimeError("invalid_novaid_signing_key")
   255	    return {"development-v1": legacy_key}, "development-v1"

    41	@contextmanager
    42	def _environment(**values: str) -> Iterator[None]:
    43	    previous = {key: os.environ.get(key) for key in values}
    44	    os.environ.update(values)
    45	    try:
    46	        yield
    47	    finally:
    48	        for key, value in previous.items():
    49	            if value is None:
    50	                os.environ.pop(key, None)
    51	            else:
    52	                os.environ[key] = value

    91	def _bootstrap(dsn: str, schema: str) -> dict[str, Any]:
    92	    with _connection(dsn, schema) as connection:
    93	        started = perf_counter()
    94	        revisions = apply_migrations(connection)
    95	        elapsed_ms = (perf_counter() - started) * 1000
    96	        verified = verify_migration_revisions(connection)
    97	        checksums = connection.execute(
    98	            "SELECT revision,checksum FROM novaid_schema_migrations ORDER BY revision"
    99	        ).fetchall()
   100	        constraints = connection.execute(
   101	            "SELECT count(*) FROM information_schema.table_constraints "
   102	            "WHERE table_schema=%s AND constraint_type IN "
   103	            "('PRIMARY KEY','FOREIGN KEY','UNIQUE','CHECK')",
   104	            (schema,),
   105	        ).fetchone()[0]
   106	        tables = connection.execute(
   107	            "SELECT count(*) FROM information_schema.tables WHERE table_schema=%s",
   108	            (schema,),
   109	        ).fetchone()[0]
   110	    if revisions != EXPECTED_REVISIONS or verified != EXPECTED_REVISIONS:
   111	        raise RuntimeError("canonical_migration_chain_not_verified")
   112	    if not all(checksum for _, checksum in checksums):
   113	        raise RuntimeError("migration_checksum_missing")
   114	    return {
   115	        "revision_count": len(revisions),
   116	        "revisions": list(revisions),
   117	        "checksums_present": True,
   118	        "table_count": tables,
   119	        "constraint_count": constraints,
   120	        "bootstrap_ms": round(elapsed_ms, 3),
   121	    }
